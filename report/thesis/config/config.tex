\renewcommand*{\bibfont}{\footnotesize}

\renewbibmacro*{volume+number+eid}{%
  \printfield{volume}%
%  \setunit*{\adddot}% DELETED
  \setunit*{\addnbthinspace}% NEW (optional); there's also \addnbspace
  \printfield{number}%
  \setunit{\addcomma\space}%
  \printfield{eid}}
\DeclareFieldFormat[article]{number}{\mkbibparens{#1}}

% remove Oxford comma in bibliography
%\renewcommand*{\finalnamedelim}{\addspace \bibstring {and}\space}

% link title to doi in bibliography
\ExecuteBibliographyOptions{doi=false}
\ExecuteBibliographyOptions{url=false}
\newbibmacro{string+doi}[1]{%
  \iffieldundef{doi}{ \iffieldundef{url}{#1}{\href{\thefield{url}}{#1}}}{\href{http://dx.doi.org/\thefield{doi}}{#1}}}
\DeclareFieldFormat{title}{\usebibmacro{string+doi}{\mkbibemph{#1}}}
\DeclareFieldFormat[article,inbook,incollection,inproceedings,patent,thesis,unpublished]{title}{\usebibmacro{string+doi}{\mkbibquote{#1\isdot}}}
