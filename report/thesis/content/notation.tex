\chapter*{Notation}
\section*{Roman Letters}
\begin{mathlist}
\item[\vct{r}] vector
\item[\matr{A}] matrix
\item[a_{ij}] entry $(i,j)$ of matrix $\matr{A}$
\end{mathlist}

\section*{Other Mathematical Symbols}
\begin{mathlist}
\item[\|\cdot\|_1] $\ell_1$ norm: $\|\vct{r}\|_1 = \sum_{i} |r_i|$
\item[\|\cdot\|_2] $\ell_2$ (euclidean) norm: $\|\vct{r}\|_2 = \sqrt{\sum_{i} |r_i|^2}$
\end{mathlist}

\section*{Physical Quantities}
\begin{mathlist}
\item[\vct{G}] Global descriptor of the system for a given frame.
\item[\matr{G}] Matrix of global descriptors for a collection of frames.
\item[\matr{V}_{PCA}] PCA projection matrix.
\item[\vct{\mu}] Mean of the data used for PCA. 
\item[\vct{m}] Midpoint of the products cluster.
\end{mathlist}
